Sintaxe Linguagem C++ e Programação Orientada a Objetos

Estrutura Básica:

#include <iostream>
using namespace std;

int main() {
	// Comandos do programa principal
	return 0;
}

Entrada e Saída Padrão (iostream):

cout << dado << endl;

dado: valor ou variável a ser exibida no console (endl adiciona quebra de linha).

cin >> variável ;

variável: variável que receberá o dado digitado via teclado.

getline(cin, stringVar);

stringVar: variável do tipo string para ler uma linha completa com espaços.

Limpeza do Buffer de Entrada:

cin.ignore();

Descarta um único caractere do buffer (útil para remover o '\n' restante de uma leitura anterior).

cin.ignore(10000, '\n');

Limpa todos os caracteres pendentes no buffer até encontrar a quebra de linha ('\n').

Tipos de Dados:

int float double char bool string void

Declaração de Variáveis:

tipo nome ;
tipo nome = valorInicial ;

tipo: tipo de dado;

nome: nome da variável.

Atribuição:

variável = expressão ;

Definição de Constantes:

const int nome = valor ;

nome: nome da constante;

valor: valor da constante.

#define NOME VALOR

NOME: nome da constante;

VALOR: valor da constante.

Operadores:

+  -  *  /  %

==  !=  >  >=  <  <=

!  &&  ||

++   --

Estruturas de Decisão:

if (condição1) {
	comandos1
}
else if (condição2) {
	comandos2
}
else {
	comandos padrão
}



switch(variável) {
	case constante1:
		comandos1
		break;

	case constante2:
		comandos2
		break;

	default:
		comandos padrão
		break;
}

variável: a variável a ser comparada com as constantes;

constante1, constante2, …: as constantes a serem comparadas com a variável.

Estruturas de Repetição:

for(var = ini; var <= fim; var++) {
	trecho de repetição
}

var: variável de contagem;

ini: valor inicial para a contagem;

fim: valor final para a contagem.

while(condição) {
	trecho de repetição
}



do {
	trecho de repetição
} while(condição);

Declaração de Vetores de Uma Dimensão:

tipo nome [tamanho];
tipo nome [tamanho] = { valor0, valor1, ... } ;

Acesso e Atribuição em Vetores:

nome [ índice ] = expressão ;
variável = nome [ índice ] ;

índice: posição do elemento no vetor (varia de 0 até tamanho - 1).

Definição de Função com Retorno de Valor:

tipo nomeDaFuncao(argumento1, argumento2, ...) {
	comandos
	return valor;
}

Definição de Função sem Retorno de Valor:

void nomeDaFuncao(argumento1, argumento2, ...) {
	comandos
}

Protótipo de Função:

tipo nomeDaFuncao(argumento1, argumento2, ...);
void nomeDaFuncao(argumento1, argumento2, ...);

Estrutura Completa de Definição de Classe:

class NomeDaClasse {
private:
	tipo atributoPrivado;

protected:
	tipo atributoProtegido;

public:
	// Construtor Padrão
	NomeDaClasse() {
		atributoPrivado = 0;
	}

	// Construtor com Parâmetros
	NomeDaClasse(tipo valor) {
		atributoPrivado = valor;
	}

	// Destrutor
	~NomeDaClasse() {
		// Liberação de recursos, se necessário
	}

	// Método sem retorno (void)
	void metodo1(tipo parametro) {
		// Comandos do método sem retorno
	}

	// Método com retorno de valor
	tipo metodo2() {
		// Comandos do método com retorno
		return atributoPrivado;
	}

	// Protótipo de Método (para implementação externa)
	void metodoExterno();
};

private: visibilidade privada (acesso restrito aos membros da própria classe);

protected: visibilidade protegida (acesso permitido à própria classe e subclasses);

public: visibilidade pública (acesso livre externamente);

NomeDaClasse() / NomeDaClasse(valor): construtores (executados ao instanciar o objeto);

~NomeDaClasse(): destrutor (executado ao destruir o objeto);

metodo1 / metodo2: métodos da classe (um do tipo void e outro com retorno de valor).

Implementação de Método Fora da Classe:

void NomeDaClasse::metodoExterno() {
	// Código do método definido fora do corpo da classe
}

:: (operador de resolução de escopo): vincula a definição externa da função à classe correspondente.

Instanciação de Objetos e Acesso a Membros:

// Instanciação usando Construtor Padrão
NomeDaClasse objeto1;

// Instanciação usando Construtor com Parâmetros
NomeDaClasse objeto2(valorInicial);

// Chamada de Métodos Públicos
objeto1.metodo1(arg);
variável = objeto1.metodo2();
objeto1.metodoExterno();

objeto1 / objeto2: instâncias criadas a partir da classe;

. (ponto): operador para acessar os métodos e atributos públicos de um objeto.
